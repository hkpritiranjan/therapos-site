Technical

Database Schema

PostgreSQL 15 via Prisma ORM with the pgvector extension for embedding storage. All tables use CUID primary keys. PHI fields are additionally encrypted at the application layer via envelope encryption.

Entity Relationship Map

User ──────────────────── Patient ──── Session ──── SessionNote
 │                          │              │              │
 │                          │              │         NoteEmbedding
 │                          │              │
 │                          ├── Goal       └── Homework
 │                          ├── Homework
 │                          └── PatientEmbedding
 │
 ├── Subscription
 └── AuditLog

Key Design Decisions

  • CUIDs over UUIDs: CUIDs are URL-safe by default, have better lexicographic properties than UUIDs v4, and sort naturally in most UI contexts without additional logic.
  • Soft deletes on patients: Patients are never hard-deleted — HIPAA audit requirements mean we must retain records. Deletion sets status = DISCHARGED and endDate. PHI is removed on account cancellation per our data retention policy.
  • JSON for flexible data: Session.aiPrep and Goal.progressNotes are JSON columns. This avoids schema migrations for structured data that evolves with the AI output format. The JSON shape is validated in application code via Zod.
  • Separate embedding tables: NoteEmbedding and PatientEmbedding are separate tables rather than vector columns on the parent table. This keeps the main tables lean and allows independent indexing strategy for the vector data.
  • therapistId on every leaf table: Sessions, Goals, Homework all carry a therapistId foreign key even though it's derivable through the Patient. This enables fast therapist-scoped queries without table joins, and is critical for the pgvector search query where we filter embeddings by therapist before similarity ranking.

// ── User ──────────────────────────────────────────────────────

model User {
  id            String        @id @default(cuid())
  clerkId       String        @unique  // Clerk's user ID
  email         String        @unique
  name          String
  licenseNumber String?
  licenseState  String?       // US state abbreviation
  credentials   String[]      // ["LCSW", "MFT", "PhD", "LPC"]
  modalities    String[]      // ["CBT", "DBT", "Psychodynamic"]
  timezone      String        @default("UTC")
  subscription  Subscription?
  patients      Patient[]
  sessions      Session[]
  auditLogs     AuditLog[]
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
}

// ── Patient ───────────────────────────────────────────────────

model Patient {
  id              String            @id @default(cuid())
  therapistId     String
  therapist       User              @relation(fields: [therapistId], references: [id])
  // PHI — application-layer encrypted
  firstName       String
  lastName        String
  preferredName   String?
  dateOfBirth     DateTime?
  email           String?
  phone           String?
  // Clinical
  primaryConcerns String[]          // ["anxiety", "workplace stress"]
  diagnoses       String[]          // ["F41.1 GAD", "F43.10 PTSD"]
  modality        String?           // Per-patient override of therapist default
  sessionFreq     String?           // "weekly" | "biweekly" | "monthly"
  status          PatientStatus     @default(ACTIVE)
  startDate       DateTime          @default(now())
  endDate         DateTime?
  // AI-maintained — updated after each finalized note via BullMQ
  aiSummary          String?
  aiSummaryUpdatedAt DateTime?
  // Relations
  sessions        Session[]
  goals           Goal[]
  homework        Homework[]
  embeddings      PatientEmbedding[]
  createdAt       DateTime          @default(now())
  updatedAt       DateTime          @updatedAt

  @@index([therapistId])
  @@index([therapistId, status])    // Dashboard: active patients only
}

enum PatientStatus {
  ACTIVE
  INACTIVE
  WAITLIST
  DISCHARGED
}

// ── Session ───────────────────────────────────────────────────

model Session {
  id            String        @id @default(cuid())
  patientId     String
  patient       Patient       @relation(fields: [patientId], references: [id])
  therapistId   String        // Denormalized for fast therapist-scoped queries
  therapist     User          @relation(fields: [therapistId], references: [id])
  sessionNumber Int           // Auto-incremented per patient
  scheduledAt   DateTime
  startedAt     DateTime?
  endedAt       DateTime?
  durationMins  Int?
  type          SessionType   @default(INDIVIDUAL)
  status        SessionStatus @default(SCHEDULED)
  // AI prep brief — JSON: { recap, suggestedQuestions[], homeworkFollowUp, patterns }
  aiPrep        Json?
  aiPrepGenAt   DateTime?
  // Therapist's raw notes before AI processing
  rawNotes      String?
  note          SessionNote?
  homework      Homework[]
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt

  @@index([patientId])
  @@index([therapistId])
  @@index([therapistId, scheduledAt])  // Dashboard: today's sessions
  @@unique([patientId, sessionNumber]) // Enforce unique session numbers per patient
}

enum SessionType {
  INDIVIDUAL
  COUPLES
  FAMILY
  GROUP
}

enum SessionStatus {
  SCHEDULED
  IN_PROGRESS
  COMPLETED
  CANCELLED
  NO_SHOW
}

// ── SessionNote ───────────────────────────────────────────────

model SessionNote {
  id            String     @id @default(cuid())
  sessionId     String     @unique  // One note per session
  session       Session    @relation(fields: [sessionId], references: [id])
  format        NoteFormat @default(SOAP)
  // SOAP fields
  subjective    String?
  objective     String?
  assessment    String?
  plan          String?
  // DAP fields
  data          String?
  dapAssessment String?
  dapPlan       String?
  // BIRP fields
  behavior      String?
  intervention  String?
  birpResponse  String?
  birpPlan      String?
  // Free text
  freeText      String?
  // AI tracking — never modify aiDraft after finalization
  aiDraft       Json?      // Original AI output before therapist edits
  aiGenerated   Boolean    @default(false)
  aiModel       String?    // e.g. "claude-sonnet-4-6"
  editedAt      DateTime?  // Last manual edit timestamp
  isFinalized   Boolean    @default(false)
  finalizedAt   DateTime?
  wordCount     Int?       // Computed on finalization
  // Relations
  embedding     NoteEmbedding?
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt
}

enum NoteFormat {
  SOAP
  DAP
  BIRP
  FREE
}

// ── Goal ─────────────────────────────────────────────────────

model Goal {
  id            String     @id @default(cuid())
  patientId     String
  patient       Patient    @relation(fields: [patientId], references: [id])
  title         String
  description   String?
  category      String?    // "anxiety" | "sleep" | "relationships" | "trauma" | ...
  status        GoalStatus @default(ACTIVE)
  startedAt     DateTime   @default(now())
  achievedAt    DateTime?
  // Array of { date: ISO, sessionId?: string, note: string }
  progressNotes Json[]     @default([])
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt

  @@index([patientId])
  @@index([patientId, status])
}

enum GoalStatus {
  ACTIVE
  ACHIEVED
  PAUSED
  ABANDONED
}

// ── Homework ──────────────────────────────────────────────────

model Homework {
  id          String         @id @default(cuid())
  patientId   String
  patient     Patient        @relation(fields: [patientId], references: [id])
  sessionId   String?        // Session when assigned (nullable for manually added homework)
  session     Session?       @relation(fields: [sessionId], references: [id])
  title       String
  description String
  category    String?        // "journaling" | "behavioral" | "exposure" | "mindfulness" | "social"
  dueBy       DateTime?
  aiGenerated Boolean        @default(false)
  status      HomeworkStatus @default(ASSIGNED)
  completedAt DateTime?
  notes       String?        // Therapist's completion notes
  createdAt   DateTime       @default(now())
  updatedAt   DateTime       @updatedAt

  @@index([patientId])
  @@index([patientId, status])
}

enum HomeworkStatus {
  ASSIGNED
  COMPLETED
  PARTIAL
  SKIPPED
}

// ── Vector Tables ──────────────────────────────────────────────

model NoteEmbedding {
  id          String      @id @default(cuid())
  noteId      String      @unique
  note        SessionNote @relation(fields: [noteId], references: [id])
  patientId   String      // Denormalized for fast patient-scoped search
  therapistId String      // Denormalized for therapist isolation in queries
  content     String      // The text that was embedded (all note fields concatenated)
  // pgvector column — 1536 dimensions for text-embedding-3-small
  embedding   Unsupported("vector(1536)")
  createdAt   DateTime    @default(now())

  @@index([patientId])
  @@index([therapistId])
}

model PatientEmbedding {
  id        String   @id @default(cuid())
  patientId String
  patient   Patient  @relation(fields: [patientId], references: [id])
  type      String   // "summary" | "goal" | "pattern"
  content   String
  embedding Unsupported("vector(1536)")
  createdAt DateTime @default(now())

  @@index([patientId])
}

// ── Subscription ──────────────────────────────────────────────

model Subscription {
  id                 String    @id @default(cuid())
  userId             String    @unique
  user               User      @relation(fields: [userId], references: [id])
  plan               Plan      @default(TRIAL)
  status             SubStatus @default(TRIALING)
  stripeCustomerId   String?
  stripeSubId        String?
  trialEndsAt        DateTime?
  currentPeriodStart DateTime?
  currentPeriodEnd   DateTime?
  cancelAt           DateTime?  // Scheduled cancellation date
  cancelledAt        DateTime?  // Actual cancellation date
  pausedAt           DateTime?
  pauseResumesAt     DateTime?
  // India-specific
  razorpayCustomerId String?
  razorpaySubId      String?
  currency           String    @default("USD")  // "USD" | "INR"
  createdAt          DateTime  @default(now())
  updatedAt          DateTime  @updatedAt
}

enum Plan {
  TRIAL
  SOLO
  CLINIC
}

enum SubStatus {
  TRIALING
  ACTIVE
  PAST_DUE
  PAUSED
  CANCELLED
}

// ── HIPAA Audit Log ───────────────────────────────────────────

model AuditLog {
  id         String   @id @default(cuid())
  userId     String
  user       User     @relation(fields: [userId], references: [id])
  patientId  String?  // Null for non-patient actions (login, settings)
  action     String   // "PATIENT_VIEWED" | "NOTE_CREATED" | "AI_GENERATED" | "NOTE_FINALIZED" | ...
  resource   String   // "Patient" | "Session" | "SessionNote" | "Goal" | ...
  resourceId String
  metadata   Json?    // Non-PHI context: format chosen, note word count, etc.
  ipAddress  String?
  userAgent  String?
  createdAt  DateTime @default(now())

  @@index([userId])
  @@index([patientId])
  @@index([createdAt])  // Range queries for compliance reports
}

pgvector Setup

Run these SQL commands once, after Prisma migrations have created the tables. These are not Prisma migrations — run them directly via psql or a migration script.

-- Enable the pgvector extension
CREATE EXTENSION IF NOT EXISTS vector;

-- HNSW index on NoteEmbedding — best recall/latency for <1M vectors
-- m=16: number of bi-directional links per node (higher = better recall, more memory)
-- ef_construction=64: size of the candidate list during index build (higher = better quality)
CREATE INDEX note_embedding_hnsw_idx
ON "NoteEmbedding"
USING hnsw (embedding vector_cosine_ops)
WITH (m = 16, ef_construction = 64);

-- Set the search quality parameter at query time (higher = more accurate, slower)
SET hnsw.ef_search = 40;

Vector Search Query

The NL search endpoint embeds the user's query and runs this SQL (parameterized). The therapistId filter ensures therapist isolation before similarity ranking.

-- Natural language search — therapist-scoped, optionally patient-scoped
-- $1 = query embedding (vector), $2 = therapistId, $3 = patientId (nullable)
SELECT
  ne.content,
  ne."patientId",
  ne."noteId",
  s."sessionId",
  s."scheduledAt",
  p."firstName",
  p."lastName",
  sn."format",
  1 - (ne.embedding <=> $1::vector) AS score
FROM "NoteEmbedding" ne
JOIN "SessionNote" sn  ON sn.id = ne."noteId"
JOIN "Session"    s    ON s.id = sn."sessionId"
JOIN "Patient"    p    ON p.id = ne."patientId"
WHERE ne."therapistId" = $2
  AND ($3::text IS NULL OR ne."patientId" = $3)
ORDER BY ne.embedding <=> $1::vector
LIMIT 10;

-- The <=> operator is cosine distance (lower = more similar)
-- 1 - distance = similarity score (higher = more similar)
-- Filtered by therapistId before vector scan — pgvector uses the HNSW index
